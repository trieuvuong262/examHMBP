(function () { var form = document.getElementById('npl-import-form'); var drop = document.getElementById('npl-import-drop'); var input = document.getElementById('npl-excel-file'); var submit = document.getElementById('npl-import-submit'); var meta = document.getElementById('npl-import-file-meta'); var modal = document.getElementById('nplImportModal'); if (!form || !drop || !input || !submit) return; var MAX_MB = 10; function formatSize(bytes) { if (bytes < 1024) return bytes + ' B'; if (bytes < 1024 * 1024) return (bytes / 1024).toFixed(1) + ' KB'; return (bytes / (1024 * 1024)).toFixed(1) + ' MB'; } function isExcel(name) { var lower = (name || '').toLowerCase(); return lower.endsWith('.xlsx') || lower.endsWith('.xls'); } function setFile(file) { if (!file) { input.value = ''; drop.classList.remove('has-file'); meta.classList.add('d-none'); meta.textContent = ''; submit.disabled = true; return; } if (!isExcel(file.name)) { alert('Chỉ chấp nhận file Excel (.xlsx hoặc .xls).'); setFile(null); return; } if (file.size > MAX_MB * 1024 * 1024) { alert('File quá lớn. Tối đa ' + MAX_MB + ' MB.'); setFile(null); return; } var dt = new DataTransfer(); dt.items.add(file); input.files = dt.files; drop.classList.add('has-file'); meta.classList.remove('d-none'); meta.textContent = file.name + ' (' + formatSize(file.size) + ')'; submit.disabled = false; } input.addEventListener('change', function () { setFile(input.files && input.files[0] ? input.files[0] : null); }); ['dragenter', 'dragover'].forEach(function (ev) { drop.addEventListener(ev, function (e) { e.preventDefault(); drop.classList.add('is-dragover'); }); }); ['dragleave', 'drop'].forEach(function (ev) { drop.addEventListener(ev, function (e) { e.preventDefault(); drop.classList.remove('is-dragover'); }); }); drop.addEventListener('drop', function (e) { var file = e.dataTransfer && e.dataTransfer.files && e.dataTransfer.files[0]; if (file) setFile(file); }); form.addEventListener('submit', function () { submit.disabled = true; submit.querySelector('.jp-npl-import-submit-label').classList.add('d-none'); submit.querySelector('.jp-npl-import-submit-loading').classList.remove('d-none'); }); if (modal) { modal.addEventListener('hidden.bs.modal', function () { setFile(null); submit.querySelector('.jp-npl-import-submit-label').classList.remove('d-none'); submit.querySelector('.jp-npl-import-submit-loading').classList.add('d-none'); }); } })();
